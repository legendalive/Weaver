# weaver-server

All server-side code for Weaver (Supabase Edge Functions).
Companion repo: `legendalive/Weaver` (the app + admin UI).

## Architecture (V3.1a pivot, 2026-10-24)
Supabase Edge Functions CANNOT serve HTML: their gateway forces a
`sandbox` Content-Security-Policy without `allow-scripts`, so any HTML a
function returns renders as dead text in the browser. Therefore:

| Component | Where | What |
|---|---|---|
| Admin **UI** | `Weaver` repo → `admin/index.html` → GitHub Pages | branded dashboard, served as a normal webpage |
| Admin **API** | this repo → `supabase/functions/admin` | JSON only; owner-JWT gated |

The UI calls the API cross-origin; the function whitelists
`https://legendalive.github.io` via CORS. Never widen that origin
without a written reason here.

- **Admin console:** https://legendalive.github.io/Weaver/admin/
- **API base:** https://hiqmvhkwuvvgalqxxtbc.functions.supabase.co/admin
  (`/api/health` open · `/api/overview` owner-only)

## Deploy
Automatic: push to `main` → GitHub Action deploys every function.
Requires repo secret `SUPABASE_ACCESS_TOKEN` (already set).

## Manual values registry
| Value | Where | Status |
|---|---|---|
| SUPABASE_ACCESS_TOKEN | repo Settings → Secrets → Actions | ✅ set |
| Project ref `hiqmvhkwuvvgalqxxtbc` | deploy-functions.yml | ✅ pre-filled |
| `OWNER_USER_ID` | top of `supabase/functions/admin/index.ts` | ⚠️ paste your User UID (Supabase → Authentication → Users → your account) |
| Supabase URL / anon / service_role | auto-injected at runtime | nothing to do |

## Rules
- `service_role` must NEVER appear in this repo or any client file.
- Functions return JSON only — no HTML (the CSP sandbox kills it).
- Every delivered file states its exact path; every step lists its manual pastes.

## Troubleshooting
- Action fails → open the run → first red step names the cause.
- API 404 → deploy hasn't finished; check the Actions tab.
- "unauthorized" after sign-in → `OWNER_USER_ID` doesn't match your User UID.
- Dashboard says "Could not reach the admin API" → function not deployed,
  or the CORS origin no longer matches the Pages URL.

  try {
    const users = await fetchAllUsers();
    const now = Date.now(); const day = 86400000;
    out.usersTotal = users.length;
    out.users7d = users.filter((u) => now - Date.parse(u.created_at) <= 7 * day).length;
    out.users30d = users.filter((u) => now - Date.parse(u.created_at) <= 30 * day).length;
  } catch (e) { (out.problems as string[]).push('users: ' + (e as Error).message); }
  try {
    out.objects = await countStorageObjects();
    if (out.objects === null) (out.problems as string[]).push('storage count unavailable');
  } catch (e) { (out.problems as string[]).push('storage count: ' + (e as Error).message); }
  try { out.buckets = await listBuckets(); }
  catch (e) { (out.problems as string[]).push('buckets: ' + (e as Error).message); }
  return out;

async function adjustCredits(userId: string, body: any) {
  const amount = parseInt(String(body.amount), 10) || 0;
  const reason = String(body.reason || '').trim();
  const proDays = parseInt(String(body.pro_days || 0), 10) || 0;
  if (amount === 0 && proDays <= 0) throw new Error('enter a credit amount and/or Pro days');
  if (!reason) throw new Error('a reason is required (it goes into the audit log)');
  const current = await getEntitlement(userId);
  const currentCredits = current ? (current.credits || 0) : 0;
  const newCredits = Math.max(0, currentCredits + amount);
  let proUntil: string | null = current ? current.pro_until : null;
  if (proDays > 0) {
    const base = proUntil && Date.parse(proUntil) > Date.now() ? Date.parse(proUntil) : Date.now();
    proUntil = new Date(base + proDays * 86400000).toISOString();
  }
  const patch: Record<string, unknown> = {};
  if (amount !== 0) patch.credits = newCredits;
  if (proUntil) patch.pro_until = proUntil;
  if (!Object.keys(patch).length) throw new Error('nothing to change');
  const row = await upsertEntitlement(userId, patch);
  const bits: string[] = [];
  if (amount !== 0) bits.push(`${amount > 0 ? '+' : ''}${amount} credits`);
  if (proDays > 0) bits.push(`+${proDays}d Pro`);
  const summary = bits.join(', ');
  await addCreditLog(userId, amount, `${reason} (${summary})`);
  await logAdmin('credits.adjust', await emailFor(userId), `${summary} — ${reason}`);
  return row;
}

async function listVaultFiles(): Promise<{ path: string; owner: string; size: number; updated_at: string }[]> {
  const files: { path: string; owner: string; size: number; updated_at: string }[] = [];
  const queue: string[] = [''];
  let guard = 0;
  while (queue.length > 0 && guard < 500) {
    const prefix = queue.shift() as string;
    let offset = 0;
    for (;;) {
      guard++;
      if (guard > 500) break;
      const res = await fetch(`${SB_URL}/storage/v1/object/list/weaver-vault`, {
        method: 'POST',
        headers: { ...adminHeaders(), 'Content-Type': 'application/json' },
        body: JSON.stringify({ prefix, limit: 1000, offset }),
      });
      if (!res.ok) {
        const t = await res.text();
        throw new Error(`storage list HTTP ${res.status} ${t.slice(0, 160)}`);
      }
      const batch = await res.json();
      if (!Array.isArray(batch) || batch.length === 0) break;
      for (const item of batch) {
        const name = String(item.name || '');
        if (!name) continue;
        const full = prefix && !name.startsWith(prefix) ? prefix + name : name;
        const size = item.metadata && item.metadata.size;
        if (size == null) {
          queue.push(full + '/');
        } else {
          const sz = typeof size === 'number' ? size : (parseInt(size, 10) || 0);
          files.push({ path: full, owner: full.split('/')[0] || '', size: sz, updated_at: item.updated_at || '' });
        }
      }
      if (batch.length < 1000) break;
      offset += 1000;
    }
  }
  return files;
}
async function storageReport() {
  const files = await listVaultFiles();
  const byUser: Record<string, { objects: number; bytes: number; projects: number; last: string }> = {};
  let totalBytes = 0;
  for (const f of files) {
    const uid = f.owner;
    if (!uid) continue;
    const rec = byUser[uid] || (byUser[uid] = { objects: 0, bytes: 0, projects: 0, last: '' });
    rec.objects++;
    rec.bytes += f.size;
    totalBytes += f.size;
    if (f.path.indexOf('/weaver/projects/') >= 0 && f.path.endsWith('.json')) rec.projects++;
    if (f.updated_at && f.updated_at > rec.last) rec.last = f.updated_at;
  }
  const users = await fetchAllUsers();
  const emailById: Record<string, string> = {};
  for (const u of users) emailById[u.id] = u.email;
  const perUser = Object.entries(byUser).map(([uid, r]) => ({
    user_id: uid,
    email: emailById[uid] || '(deleted account)',
    objects: r.objects,
    bytes: r.bytes,
    projects: r.projects,
    last: r.last,
  })).sort((a, b) => b.bytes - a.bytes);
  return { totalObjects: files.length, totalBytes, perUser };
}
/* =========================================================
PROJECT-LEVEL ADMIN (V3.1i)
Privacy model: reads the vault index (project TITLES) and the
storage listing (SIZES) only. Project blobs are deleted by
path — never downloaded, opened, or logged. Manuscript
content never crosses into the admin console.
========================================================= */
async function decodeVaultJson(bytes: Uint8Array): Promise<any> {
  let text: string;
  if (bytes.length > 2 && bytes[0] === 0x1f && bytes[1] === 0x8b) {
    const stream = new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'));
    text = await new Response(stream).text();
  } else {
    text = new TextDecoder().decode(bytes);
  }
  return JSON.parse(text);
}
async function readUserIndex(userId: string): Promise<any | null> {
  const res = await fetch(`${SB_URL}/storage/v1/object/weaver-vault/${userId}/weaver/index.json`, {
    headers: adminHeaders(),
  });
  if (res.status === 404 || res.status === 400) return null;
  if (!res.ok) throw new Error(`read index HTTP ${res.status}`);
  const bytes = new Uint8Array(await res.arrayBuffer());
  return await decodeVaultJson(bytes);
}
async function listUserProjects(userId: string) {
  const idx = await readUserIndex(userId);
  if (!idx) return { neverSynced: true, projects: [] };
  const files = await listVaultFiles();
  const sizeByPath: Record<string, number> = {};
  for (const f of files) {
    if (f.owner === userId) sizeByPath[f.path] = f.size;
  }
  const projects: any[] = [];
  for (const [pid, info] of Object.entries((idx.projects || {}) as Record<string, any>)) {
    const path = `${userId}/weaver/projects/${pid}.json`;
    projects.push({
      id: pid,
      name: (info && info.name) || '(untitled)',
      updatedAt: (info && info.updatedAt) || null,
      bytes: sizeByPath[path] || 0,
      missing: !(path in sizeByPath),
    });
  }
  projects.sort((a, b) => (b.updatedAt || 0) - (a.updatedAt || 0));
  return { neverSynced: false, projects };
}
async function deleteUserProject(userId: string, projectId: string) {
  if (!/^[A-Za-z0-9_-]+$/.test(projectId)) throw new Error('invalid project id');
  const idx = await readUserIndex(userId);
  if (!idx || !idx.projects || !idx.projects[projectId]) {
    throw new Error('project not found in the vault index');
  }
  const name = idx.projects[projectId].name || projectId;
  /* 1) delete the blob by path — its content is never fetched */
  const del = await fetch(`${SB_URL}/storage/v1/object/weaver-vault/${userId}/weaver/projects/${projectId}.json`, {
    method: 'DELETE',
    headers: adminHeaders(),
  });
  if (!del.ok && del.status !== 404 && del.status !== 400) {
    throw new Error(`delete blob HTTP ${del.status}`);
  }
  /* 2) rewrite index.json without the entry (raw JSON — the app
        sniffs gzip magic bytes and reads plain JSON fine) */
  delete idx.projects[projectId];
  idx.updatedAt = Date.now();
  const up = await fetch(`${SB_URL}/storage/v1/object/weaver-vault/${userId}/weaver/index.json`, {
    method: 'POST',
    headers: { ...adminHeaders(), 'x-upsert': 'true', 'Content-Type': 'application/json' },
    body: JSON.stringify(idx),
  });
  if (!up.ok) {
    const t = await up.text();
    throw new Error(`rewrite index HTTP ${up.status} ${t.slice(0, 120)}`);
  }
  return name;
}
async function metrics() {
  const users = await fetchAllUsers();
  const now = Date.now();
  const day = 86400000;
  const keys: string[] = [];
  const buckets: Record<string, number> = {};
  for (let i = 13; i >= 0; i--) {
    const key = new Date(now - i * day).toISOString().slice(0, 10);
    keys.push(key);
    buckets[key] = 0;
  }
  let active7d = 0;
  let active30d = 0;
  for (const u of users) {
    const created = String(u.created_at || '').slice(0, 10);
    if (created in buckets) buckets[created]++;
    if (u.last_sign_in_at) {
      const t = Date.parse(u.last_sign_in_at);
      if (Number.isFinite(t)) {
        if (now - t <= 7 * day) active7d++;
        if (now - t <= 30 * day) active30d++;
      }
    }
  }
  return {
    usersTotal: users.length,
    active7d,
    active30d,
    signups14d: keys.reduce((s, k) => s + buckets[k], 0),
    signups: keys.map((k) => ({ label: k.slice(5), count: buckets[k] })),
  };
}

async function getAnnouncement() {
  const res = await fetch(`${REST}/announcement?id=eq.1&select=*`, { headers: restHeaders() });
  if (!res.ok) throw new Error(`read announcement HTTP ${res.status}`);

async function saveAnnouncement(body: any) {
  const text = String(body.text || '').trim();
  const level = String(body.level || 'info');
  const showUntil = body.show_until || null;
  if (!['info', 'warning', 'maintenance'].includes(level)) throw new Error('invalid level');
  if (text.length > 400) throw new Error('banner text must be 400 characters or fewer');
  const res = await fetch(`${REST}/announcement?id=eq.1`, {
    method: 'PATCH',
    headers: restHeaders({ Prefer: 'return=representation' }),
    body: JSON.stringify({ text, level, show_until: showUntil, updated_at: new Date().toISOString() }),
  });
  if (!res.ok) {
    const t = await res.text();
    throw new Error(`save announcement HTTP ${res.status} ${t}`);
  }
  const rows = await res.json();
  return rows.length ? rows[0] : null;
}

function toBase64(s: string): string {
  const bytes = new TextEncoder().encode(s);
  let bin = '';
  for (const b of bytes) bin += String.fromCharCode(b);
  return btoa(bin);
}
function csv(rows: string[][]): string {
  return rows.map((r) => r.map((cell) => '"' + String(cell == null ? '' : cell).replace(/"/g, '""') + '"').join(',')).join('\n');
}

type TestResult = { ok: boolean; code: string; message: string; detail?: string };

const EXCLUDE_MODEL = /whisper|embed|audio|tts|vision|image|clip|rerank|guard|transcri|speech|moderation/i;

function classify(status: number, detail: string): TestResult {
  const d = (detail || '').slice(0, 220);
  if (status === 401 || status === 403) return { ok: false, code: 'invalid_key', message: 'Invalid or revoked API key (or key belongs to a different account/org).', detail: d };
  if (status === 402) return { ok: false, code: 'billing', message: 'Billing/quota depleted — the provider requires payment or more credits.', detail: d };
  if (status === 429) return { ok: false, code: 'rate_limit', message: 'Rate limit or quota window exceeded (too many requests). Wait a minute and retry.', detail: d };
  if (status === 404) return { ok: false, code: 'model_retired', message: 'Model not found / retired, or this key has no access to it.', detail: d };
  if (status >= 500) return { ok: false, code: 'provider_down', message: 'Provider server error — outage or overload on their side.', detail: d };
  return { ok: false, code: 'http_' + status, message: 'Unexpected HTTP ' + status + ' from provider.', detail: d };
}

function netErr(e: any): TestResult {
  const msg = (e && e.message) || String(e);
  if (/abort|timeout/i.test(msg)) return { ok: false, code: 'timeout', message: 'Request timed out — provider unreachable or very slow (network/traffic).', detail: msg };
  return { ok: false, code: 'network', message: 'Network error reaching the provider (connectivity/DNS/firewall).', detail: msg };
}

let docCache: { at: number; sections: { title: string; text: string }[] } | null = null;
async function loadSections() {
  if (docCache && Date.now() - docCache.at < 600000) return docCache.sections;
  let text = '';
  let lastErr = '';
  for (const url of DOC_URLS) {
    try {
      const res = await fetch(url, { cache: 'no-store' });
      if (!res.ok) { lastErr = `${url} -> HTTP ${res.status}`; continue; }
      text = await res.text();
      if (text && text.length > 200) break;
      lastErr = `${url} -> empty body`;
      text = '';
    } catch (e) { lastErr = `${url} -> ${(e as Error).message}`; }
  }
  if (!text) throw new Error('doc fetch failed: ' + lastErr);
  const sections: { title: string; text: string }[] = [];
  let cur: { title: string; parts: string[] } | null = null;
  for (const raw of text.replace(/\r\n?/g, '\n').split('\n')) {
    if (raw.startsWith('## ')) {
      if (cur) sections.push({ title: cur.title, text: cur.parts.join('\n') });
      cur = { title: raw.slice(3).trim(), parts: [] };
      continue;
    }
    if (raw.startsWith('# ')) continue;
    if (cur) cur.parts.push(raw);
  }
  if (cur) sections.push({ title: cur.title, text: cur.parts.join('\n') });
  docCache = { at: Date.now(), sections };
  return sections;
}

function retrieve(sections: { title: string; text: string }[], question: string): string {
  const stop = new Set(['the','a','an','my','i','is','are','was','were','to','of','in','on','for','and','or','it','with','when','how','do','does','did','not','no','yes','me','you','your','weaver','app','can','cannot','cant','won','about','please']);
  const toks = question.toLowerCase().replace(/[^a-z0-9\s-]/g, ' ').split(/\s+/).filter((t) => t.length > 2 && !stop.has(t));
  const scored = sections.map((sec) => {
    const hay = (sec.title + ' ' + sec.text).toLowerCase();
    let score = 0;
    for (const t of toks) {
      if (sec.title.toLowerCase().includes(t)) score += 2;
      let idx = hay.indexOf(t);
      let count = 0;
      while (idx >= 0 && count < 8) { count++; idx = hay.indexOf(t, idx + t.length); }
      score += count;
    }
    return { sec, score };
  }).filter((x) => x.score > 0).sort((a, b) => b.score - a.score);
  const picked = (scored.length ? scored : sections.slice(0, 2).map((s) => ({ sec: s, score: 0 }))).slice(0, 5);
  let notes = '';
  for (const p of picked) {
    const block = `[[ ${p.sec.title} ]]\n${p.sec.text}\n\n`;
    if (notes.length + block.length > 9000) break;
    notes += block;
  }
  return notes || '(none)';
}
